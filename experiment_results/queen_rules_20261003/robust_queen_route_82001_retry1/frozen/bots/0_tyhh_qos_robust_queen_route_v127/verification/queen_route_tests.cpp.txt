#ifndef BOT_SOURCE
#define BOT_SOURCE "../main.cpp"
#endif
#ifndef QUEEN_ROUTE
#define QUEEN_ROUTE 1
#endif
#define main bot_program_main
#include BOT_SOURCE
#undef main
#include <cassert>

Controller testController{8, Team::A, Direction::EAST, Vision{}, 64};

void put(int c, int id, bool head, int length, bool mine = true) {
    occStamp[c] = STAMP; occMine[c] = mine; occId[c] = id; occHead[c] = head;
    parts.push_back({c, id, 1, mine, head});
    if (length) idCount.push_back({id, length});
}

void setup(int queenX = 10) {
    W = H = 20; NC = W * H; HERE = cid(5, 5); LEN = 4;
    MYID = 8; MYTEAM = Team::A; ROUND = 360; POP = 4; STAMP = 1; CT = &testController;
    testController.length = LEN; testController.unit_count = POP;
    edgeT.assign(NC * 4, E_OPEN); edgePid.assign(NC * 4, -1); learnedExit.assign(NC * 4, -1);
    seenRound.assign(NC, 1); pearlDue.assign(NC, INF); pearlInfo.assign(NC, ROUND);
    visStamp.assign(NC, STAMP); occStamp.assign(NC, 0); occMine.assign(NC, 0);
    occId.assign(NC, -1); occHead.assign(NC, 0); visPearl.assign(NC, 0);
    occPart.assign(NC, 0); occFree.assign(NC, INF);
    bfsMark.assign(NC, 0); bfsDist.assign(NC, 0); bfsFirst.assign(NC, -1); bfsQ.assign(NC + 8, 0);
    parts.clear(); idCount.clear(); friendStatuses.clear(); sightings.clear(); enemyHeads.clear();
    friendGuards.clear(); visPearls.clear(); friendParts.clear();
    queenFeedFirstRound = queenFeedLastEnemy = 350;
    consolidateHoldUntil = -1; queenSeededScout = false;
    v41Target = -1; v41TargetPhase = -1; v41ExplorationMode = true; teamFrontier = 0;
    portalEnds.clear(); portalKeys.clear(); portalClaimRound.clear();
    trail = {cid(5, 8), cid(5, 7), cid(5, 6), HERE};
    for (int c : trail) put(c, MYID, c == HERE, c == HERE ? LEN : 0);
    put(cid(queenX, 5), queenId(), true, 2); put(cid(queenX, 6), queenId(), false, 0);
    chosenAction = {};
}

bool routeAction() {
#if QUEEN_ROUTE
    return queenRouteAction(testController);
#else
    return false;
#endif
}

void plan() {
    if (!queenFeedAction(testController) && !consolidateAction(testController) && !routeAction())
        takeTurn(testController);
}

int main() {
    setup(); plan();
#if QUEEN_ROUTE
    assert(chosenAction.dirs == std::vector<int>{1} && chosenAction.split == 0 && !chosenAction.donation);
    assert(queenRoute().distance == 3);
    setup(); MYID = queenId(); assert(!routeAction());
    setup(); ROUND = 279; assert(!routeAction());
    setup(); POP = 3; assert(!routeAction());
    setup(); LEN = 25; assert(!routeAction());
    setup(); queenFeedFirstRound = ROUND - 7; assert(!routeAction());
    setup(); queenFeedLastEnemy = ROUND - 7; assert(!routeAction());
    setup(); put(cid(3, 3), 3, true, 2, false); assert(!routeAction());
    setup(); visStamp[cid(10, 5)] = 0; assert(!routeAction());
    setup(); occHead[cid(10, 5)] = 0; assert(!routeAction());
    setup(); occId[cid(10, 5)] = 2; assert(!routeAction());
    setup(); parts[parts.size() - 2].id = 2; assert(!routeAction());
    setup(); std::fill(edgeT.begin(), edgeT.end(), E_UNKNOWN); assert(!routeAction());
    setup(); std::fill(edgeT.begin(), edgeT.end(), E_PORTAL); assert(!routeAction());
    setup(); for (int d = 0; d < 4; ++d) put(geo(HERE, d), 20, false, 0);
    assert(!routeAction());
    setup(); setEdge(HERE, 1, E_KELP, -1); assert(queenRoute().direction == 0);
    setup(); friendGuards.push_back({0, cid(7, 5), 2, cid(6, 5), {cid(6, 5)}, 0});
    assert(queenRoute().direction == 0);
    setup(); friendGuards.push_back({0, cid(7, 5), 2, cid(6, 5), {cid(6, 5), cid(7, 4)}, 0});
    assert(queenRoute().direction == 0);
    setup(); ROUND = 490; assert(!routeAction());
    setup(15); assert(!routeAction());
#else
    assert(chosenAction.split == 2 && chosenAction.dirs.empty());
#endif
    setup(7); plan();
    assert(chosenAction.donation && chosenAction.dirs.empty() && chosenAction.split == 0);
    std::cout << "queen_route: candidate/control rendezvous, exclusion and original feed assertions passed\n";
}
