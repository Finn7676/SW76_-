#define main gate_bot_main
#include "../main.cpp"
#undef main
#include <cassert>
#include <iostream>

void resetGate() {
    W = 20; H = 12; NC = W * H; ROUND = 30; LEN = 2; POP = 8; MYID = 3;
    HERE = cid(2, 2); STAMP = 1; TARGET = 62; MYTEAM = Team::A;
    edgeT.assign(NC * 4, E_KELP); edgePid.assign(NC * 4, -1);
    learnedExit.assign(NC * 4, -1); seenRound.assign(NC, 1);
    pearlDue.assign(NC, INF); pearlInfo.assign(NC, ROUND);
    visStamp.assign(NC, STAMP); occStamp.assign(NC, 0); occId.assign(NC, -1);
    occFree.assign(NC, INF); occPart.assign(NC, -1);
    occMine.assign(NC, 0); occHead.assign(NC, 0); visPearl.assign(NC, 0);
    bfsMark.assign(NC, 0); bfsDist.assign(NC, 0); bfsFirst.assign(NC, -1); bfsQ.assign(NC + 8, 0);
    bfsStamp = 0; portalEnds.clear(); portalKeys.clear(); gateCrossedRound.clear();
    portalClaimRound.clear(); occupiedPortalId = -1; parts.clear(); friendParts.clear();
    idCount.clear(); friendStatuses.clear(); enemyHeads.clear(); enemyHeadLen.clear();
    visPearls.clear(); trail.clear(); chosenAction = {}; teamFrontier = 0;
    v41Target = -1; v41TargetPhase = -1;
}
void room(int x0, int x1, int y0, int y1, bool resources = false) {
    for (int y = y0; y <= y1; ++y) for (int x = x0; x <= x1; ++x) {
        int c = cid(x, y);
        if (x < x1) setEdge(c, 1, E_OPEN, -1);
        if (y < y1) setEdge(c, 2, E_OPEN, -1);
        if (resources) { pearlDue[c] = ROUND; visPearl[c] = 1; visPearls.push_back(c); }
    }
}
void pairedRooms(bool resources = false) {
    room(1, 5, 1, 5, resources); room(10, 14, 1, 5);
    setEdge(HERE, 1, E_PORTAL, 17); setEdge(cid(12, 2), 1, E_PORTAL, 17);
}

int main() {
    resetGate(); pairedRooms();
    assert(portalEnds[17].size() == 2 && gateNovel(17));
    assert(gatePlan().kind == 1 && gatePlan().dir == 1);
    gateCrossedRound[17] = ROUND;
    assert(!gateNovel(17) && gatePlan().dir == -1);
    ROUND += 20;
    assert(gatePlan().dir == -1); // topology alone cannot reset explored state

    resetGate(); pairedRooms(true);
    GateRegion region = gateRegion(HERE);
    assert(region.closed && region.cells.size() == 25 && region.useful());
    assert(provenResourceRoom(HERE) == -1); // independent from the relay's 24-cell claim
    GatePlan plan = gatePlan();
    assert(plan.kind == 2 && edgeT[HERE * 4 + plan.dir] == E_OPEN);
    setEdge(cid(1, 1), 0, E_UNKNOWN, -1);
    assert(!gateRegion(HERE).closed);

    resetGate(); room(1, 3, 1, 3);
    setEdge(cid(10, 2), 1, E_PORTAL, 19);
    assert(gatePlan().dir == -1); // unknown/unconnected space is not an approach path
    setEdge(HERE, 0, E_PORTAL, 20);
    assert(portalExit(HERE, 0) == -1);
    assert(gatePlan().kind == 1 && gatePlan().dir == 0); // unknown exit terminates the plan
    portalClaimRound[20] = ROUND;
    assert(gatePlan().dir == -1);

    resetGate(); pairedRooms();
    // Both faces must be blocked: the other face is a legitimate distinct entry.
    for (int landing : {cid(12, 2), cid(13, 2)}) {
        occStamp[landing] = STAMP; occMine[landing] = 1; occId[landing] = 77;
    }
    idCount.push_back({77, 10});
    assert(gatePlan().dir == -1); // no planned entry into a longer teammate

    resetGate(); pairedRooms(true);
    ROUND = 220; assert(gatePlan().dir == -1); ROUND = 30;
    POP = 5; assert(gatePlan().dir == -1); POP = 8;
    LEN = 9; assert(gatePlan().dir == -1); LEN = 2;
    enemyHeads.push_back(cid(4, 4)); assert(gatePlan().dir == -1);

    resetGate(); pairedRooms(true); LEN = 4;
    Controller controller(MYID, Team(Team::A), Direction('S'), Vision(), 64);
    controller.length = LEN; controller.unit_count = POP; CT = &controller;
    takeTurn(controller);
    assert(chosenAction.split == 2); // harvesting preserves ordinary expansion
    resetGate(); pairedRooms(); LEN = 4; CT = &controller;
    takeTurn(controller);
    assert(chosenAction.split == 0 && chosenAction.dirs == std::vector<int>{1});
    resetGate(); pairedRooms(); LEN = 4; ROUND = 220; CT = &controller;
    takeTurn(controller);
    assert(chosenAction.split == 2); // no acquisition delay after the module horizon
    std::cout << "gate region/acquisition/bounds assertions passed\n";
}
