#ifndef BOT_SOURCE
#define BOT_SOURCE "../main.cpp"
#endif
#ifndef PORTAL_ONCE
#define PORTAL_ONCE 1
#endif
#define main bot_program_main
#include BOT_SOURCE
#undef main
#include <cassert>
#include <fstream>
#include <iostream>
#include <sstream>
#include "self_observed_cells.hpp.txt"

struct Fixture {
    Controller ct{5, Team::B, Direction::NORTH, Vision{}, 64};
    std::vector<int> window, observed;

    bool within(int c, int h) { return cheb(c, h) <= 3; }

    void put(int x, int y, int id, bool head, int release, bool mine = true) {
        int c = cid(x, y); occStamp[c] = STAMP; occId[c] = id; occMine[c] = mine;
        occHead[c] = head; occFree[c] = release; occPart[c] = int(parts.size());
        parts.push_back({c, id, 0, mine, head});
        if (mine && id != MYID) friendParts.push_back(c);
    }

    void setup(bool r39 = false, bool historical = false) {
        W = H = 32; NC = W * H; HERE = cid(4, 4); MYID = 5; MYTEAM = Team::B;
        ROUND = r39 ? 39 : 55; STAMP = 1; LEN = ct.length = 2;
        POP = ct.unit_count = r39 ? 6 : 8; TARGET = 62;
        CT = &ct; ct.head.dragon_id = MYID; ct.head.team = Team::B;
        ct.head.dir = Direction::EAST;
        edgeT.assign(NC * 4, E_UNKNOWN); edgePid.assign(NC * 4, -1);
        learnedExit.assign(NC * 4, -1); seenRound.assign(NC, 0);
        pearlDue.assign(NC, INF); pearlInfo.assign(NC, ROUND);
        visStamp.assign(NC, 0); occStamp.assign(NC, 0); occId.assign(NC, -1);
        occFree.assign(NC, INF); occPart.assign(NC, -1); occMine.assign(NC, 0);
        occHead.assign(NC, 0); visPearl.assign(NC, 0);
        bfsMark.assign(NC, 0); bfsDist.assign(NC, 0); bfsFirst.assign(NC, -1);
        bfsQ.assign(NC + 8, 0); bfsStamp = 0;
        parts.clear(); idCount.clear(); enemyHeads.clear(); enemyHeadLen.clear();
        friendParts.clear(); visPearls.clear(); friendGuards.clear(); friendStatuses.clear();
        sightings.clear(); portalEnds.clear(); portalKeys.clear(); portalClaimRound.clear();
        occupiedPortalId = -1; trail = {cid(3, 4), HERE}; trailId = MYID;
        pendingPath.clear(); pendingPortalKey = -1;
        v41Target = -1; v41TargetPhase = -1; v41ExplorationMode = true;
        v41LowFrontierRounds = 0; teamFrontier = 999; queenSeededScout = true;
        queenFeedObserver = MYID; queenFeedFirstRound = queenFeedLastEnemy = 0;
        consolidateObserver = MYID; consolidateHoldUntil = -1; chosenAction = {};
        yieldStreak = 0;
#if PORTAL_ONCE
        portalOnceObserver = -1; portalOnceCrossed.clear(); portalOnceRefresh();
#endif
        for (int c = 0; c < NC; ++c) {
            if (within(c, HERE)) visStamp[c] = STAMP;
            if (within(c, HERE) || within(c, cid(24, r39 ? 7 : 8))) seenRound[c] = 1;
        }
        if (historical) {
            std::fill(seenRound.begin(), seenRound.end(), 0);
            for (int c : r39 ? observedR39 : observedR55) seenRound[c] = 1;
        }
        std::ifstream map("../../battlecode_quickstart/maps/default.map");
        assert(map.is_open());
        std::string line;
        while (std::getline(map, line)) {
            std::istringstream row(line); std::string tag; row >> tag;
            if (tag != "EDGE") continue;
            int key, kind, pid; row >> key >> kind >> pid;
            int boundary = key / (W + 1), x = key % (W + 1);
            int c = boundary % 2 == 0 ? cid(x % W, (boundary / 2) % H) :
                                      cid(x % W, ((boundary - 1) / 2) % H);
            int d = boundary % 2 == 0 ? 0 : 3;
            if (!seenRound[c] && !seenRound[geo(c, d)]) continue;
            setEdge(c, d, kind == 0 ? E_OPEN : kind == 1 ? E_KELP : E_PORTAL, pid);
        }
        put(4, 4, 5, true, 3); put(3, 4, 5, false, 2); idCount.push_back({5, 2});
        if (r39) {
            put(5, 5, 10, true, 4); put(4, 5, 10, false, 3); put(4, 6, 10, false, 2);
            idCount.push_back({10, 3});
        }
        std::vector<int> pearls = r39 ? std::vector<int>{cid(5, 3)} :
                                      std::vector<int>{cid(4, 5), cid(6, 4), cid(6, 6)};
        for (int c : pearls) { visPearl[c] = 1; visPearls.push_back(c); pearlDue[c] = ROUND; }
    }

    void crossing(bool r39 = false) {
#if PORTAL_ONCE
        int current = HERE;
        HERE = cid(4, r39 ? 4 : 3);
        int key = cid(24, r39 ? 7 : 8) * 4 + (r39 ? 2 : 0);
        learnedExit[key] = HERE;
        portalOnceConfirm(key);
        HERE = current;
        assert(portalOnceUsed(0));
#endif
    }

    void turn() {
        if (!queenFeedAction(ct) && !consolidateAction(ct)) takeTurn(ct);
        freeHarvest(); reviewProtection(ct); reviewPortalSafety(ct); reviewQueenSurvival(ct);
    }

    void report(char const* name) {
        std::cout << name << " " << (PORTAL_ONCE ? "candidate" : "H14 control") << " action=";
        for (int d : chosenAction.dirs) std::cout << DCH[d];
        auto p = projectAction(chosenAction);
        std::cout << " split=" << chosenAction.split << " gain=" << p.len - LEN
                  << " survives=" << p.survives << " scores=";
        for (long score : moveScores) std::cout << score << ",";
        std::cout << "\n";
    }
};

int main() {
    Fixture f;
    for (bool r39 : {false, true}) {
        f.setup(r39); f.crossing(r39); f.turn(); f.report(r39 ? "r39" : "r55");
#if !PORTAL_ONCE
        assert(chosenAction.dirs == std::vector<int>{0});
#else
        if (!r39) assert(chosenAction.dirs == std::vector<int>{2});
        if (r39) assert(chosenAction.dirs == std::vector<int>{0});
#endif
    }
    for (bool r39 : {false, true}) {
        f.setup(r39, true); f.crossing(r39); f.turn();
        f.report(r39 ? "r39 self-observed history" : "r55 self-observed history");
#if PORTAL_ONCE
        assert(chosenAction.dirs == std::vector<int>{r39 ? 0 : 2});
#else
        assert(chosenAction.dirs == std::vector<int>{0});
#endif
    }
#if PORTAL_ONCE
    f.setup();
    f.turn(); assert(chosenAction.dirs == std::vector<int>{0}); // First discovery preserved.
    f.setup(); f.crossing(); f.crossing();
    assert(portalOnceCrossed == std::vector<int>{0});
    f.setup();
    portalOnceConfirm(HERE * 4 + 1);
    assert(portalOnceCrossed.empty()); // An ordinary proposal is not a crossing.
    int key = cid(24, 8) * 4;
    learnedExit[key] = cid(4, 3);
    portalOnceConfirm(key);
    assert(portalOnceCrossed.empty()); // No actual landing at that exit this turn.
    f.setup();
    portalOnceCrossed.push_back(0);
    MYID = 27;
    assert(!portalOnceRefresh() && portalOnceCrossed.empty()); // child/new ID does not inherit.
    portalOnceCrossed.push_back(0);
    assert(portalOnceUsed(0));
    assert(portalOnceRefresh() && portalOnceUsed(0));
    MYID = 5;
    assert(!portalOnceRefresh() && portalOnceCrossed.empty());
    f.setup();
    auto msg = ordinaryWire(pack(M_PORTAL, cid(4, 3), 0));
    f.ct.sonar_messages = {msg}; receive(f.ct);
    assert(!portalOnceUsed(0)); // Teammate discovery cannot authorize local history.
    f.setup(); f.crossing();
    visPearls.clear(); std::fill(visPearl.begin(), visPearl.end(), 0);
    for (int d : {1, 2, 3}) setEdge(HERE, d, E_KELP, -1);
    f.turn(); assert(chosenAction.dirs == std::vector<int>{0}); // Repeat remains legal survival.
    f.setup(); f.crossing(); ROUND = 80;
    f.turn(); assert(chosenAction.dirs == std::vector<int>{2});
    f.setup(); f.crossing();
    int portal = HERE * 4;
    edgePid[portal] = 91;
    learnedExit[portal] = cid(24, 7);
    f.turn(); assert(chosenAction.dirs == std::vector<int>{0}); // Another PID retains priority.
    f.setup(); f.crossing();
    f.ct.length = LEN = 8;
    trail = {cid(6, 6), cid(6, 5), cid(5, 5), cid(5, 4), cid(5, 3), cid(4, 3), cid(3, 4), HERE};
    assert(!portalAllowed(HERE, 0)); // Crossing history cannot relax protected landing.
    f.turn(); assert(chosenAction.dirs != std::vector<int>{0});
#else
    f.setup();
    visPearls.clear(); std::fill(visPearl.begin(), visPearl.end(), 0);
    for (int d : {1, 2, 3}) setEdge(HERE, d, E_KELP, -1);
    f.turn(); assert(chosenAction.dirs == std::vector<int>{0});
#endif
    std::cout << "portal-once source-bound assertions passed\n";
}
